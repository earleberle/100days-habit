function challengeDay(dateStr){
  if(!APP?.startDate || !dateStr) return '';
  const a=new Date(APP.startDate+'T00:00:00Z'), b=new Date(dateStr+'T00:00:00Z');
  return Math.floor((b-a)/86400000)+1;
}

  const chars=Array.from(String(text||''));
  return chars.length<=max?text:chars.slice(0,max-1).join('')+'…';
}

async function save(){
  const name=document.getElementById('person').value, targetDate=document.getElementById('targetDate').value;
  if(!name){
    startCard.classList.add('hidden');
    finalCard.classList.add('hidden');

function fileToPayload(file){
  return new Promise((resolve,reject)=>{
    if(!file.type.startsWith('image/')){reject(new Error('이미지 파일만 올릴 수 있어요.'));return}
    const reader=new FileReader();
    reader.onerror=reject;
    reader.onload=()=>{
      const img=new Image();
      img.onerror=()=>resolve({name:file.name,mimeType:file.type||'image/jpeg',data:String(reader.result).split(',')[1]});
      img.onload=()=>{
        const MAX=1800;
        let w=img.naturalWidth,h=img.naturalHeight;
        const scale=Math.min(1,MAX/Math.max(w,h)); w=Math.round(w*scale); h=Math.round(h*scale);
        const canvas=document.createElement('canvas'); canvas.width=w;canvas.height=h;
        canvas.getContext('2d').drawImage(img,0,0,w,h);
        const dataUrl=canvas.toDataURL('image/jpeg',0.82);
        resolve({name:(file.name.replace(/\.[^.]+$/,'')||'photo')+'.jpg',mimeType:'image/jpeg',data:dataUrl.split(',')[1]});
      };
      img.src=reader.result;
    };
    reader.readAsDataURL(file);
  });
}

function renderWeekly(data){
  if(!data || !Array.isArray(data.people)) return;
  const heads=data.dates.map(d=>`<div class="whead">${d.label}<br><span class="muted">${d.date.slice(5).replace('-','/')}</span></div>`).join('');
  const rows=data.people.map(p=>{
    const cells=p.cells.map((c,i)=>{
      let cls='',txt='⬜';
      if(c.beforeStart||c.afterEnd){txt='–'}
      else if(c.pass){cls='pass';txt='PASS'}
      else if(c.count===2){cls='both';txt='2/2'}
      else if(c.count===1){cls='one';txt='1/2'}
      if(c.future) cls+=' future';
      return `<button class="wcell ${cls}" type="button" ${c.future||c.beforeStart||c.afterEnd?'disabled':''} onclick="togglePass('${esc(p.name)}','${c.date}',${c.pass?'true':'false'})">${txt}</button>`;
    }).join('');
    return `<div class="wname"><div>${esc(p.name)}<div class="weeklySummary">① ${p.h1Count}/7 · ② ${p.h2Count}/7 · 패스 ${p.passCount}/2</div></div></div>${cells}`;
  }).join('');
  document.getElementById('historyContent').innerHTML=
    `<div class="notice"><b>${dot(data.weekStart)} ~ ${dot(data.weekEnd)}</b><br>셀을 누르면 해당 날짜 PASS를 사용/취소할 수 있어요.</div>
     <div class="weeklyWrap"><div class="weekly"><div class="whead">이름</div>${heads}${rows}</div></div>`;
  document.getElementById('nextWeek').disabled=!data.canNext;
}

function runConfetti(){
  const c=document.getElementById('confetti'); if(!c)return;
  const x=c.getContext('2d'),d=Math.min(devicePixelRatio||1,2),W=innerWidth,H=innerHeight;
  c.width=W*d;c.height=H*d;c.style.width=W+'px';c.style.height=H+'px';x.setTransform(d,0,0,d,0,0);
  const cs=['#ff6b6b','#ffd93d','#6bcb77','#4d96ff','#c77dff','#ff9f1c','#fff'],a=[];
  function add(n,burst){
    for(let i=0;i<n;i++)a.push({x:burst?W/2+(Math.random()-.5)*100:Math.random()*W,y:burst?H*.22:-20-Math.random()*150,
    vx:(Math.random()-.5)*(burst?12:3),vy:burst?-4-Math.random()*8:2+Math.random()*4,g:.13+Math.random()*.09,
    w:5+Math.random()*8,h:8+Math.random()*12,r:Math.random()*6.28,vr:(Math.random()-.5)*.35,c:cs[Math.floor(Math.random()*cs.length)]});
  }
  add(160,false);add(130,true);let f=0;
  function tick(){
    x.clearRect(0,0,W,H);
    for(let i=a.length-1;i>=0;i--){let p=a[i];p.vy+=p.g;p.x+=p.vx;p.y+=p.vy;p.r+=p.vr;
      x.save();x.translate(p.x,p.y);x.rotate(p.r);x.fillStyle=p.c;x.fillRect(-p.w/2,-p.h/2,p.w,p.h);x.restore();
      if(p.y>H+40)a.splice(i,1);
    }
    f++;if(f===45||f===95)add(80,false);
    if(a.length&&f<420)requestAnimationFrame(tick);else x.clearRect(0,0,W,H);
  }
  requestAnimationFrame(tick);
}
